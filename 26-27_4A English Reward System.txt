        if(confirm("Transfer current group points to all respective students and reset group scores?")) {

                    student.points += groupScores[gIdx];

        }
    }

    function addStudent() {

        const groupInput = document.getElementById('studentGroupInput');

            renderStudents();
        }

        if (confirm(`Delete student: ${students[index].name}?`)) {
            students.splice(index, 1);
            saveStudents();
            renderStudents();
        }
    }

    function saveStudents() {
        localStorage.setItem('comicStudents', JSON.stringify(students));
    }

    // --- Rewards Logic ---
    function renderRewards() {
        const container = document.getElementById('rewardsContainer');
        container.innerHTML = '';

        rewardsList.forEach(reward => {
            const card = document.createElement('div');
            card.className = 'reward-card';
            card.innerHTML = `
                <div>
                    <div class="reward-icon">${reward.icon}</div>
                    <div class="reward-title">${reward.title}</div>
                    <div class="reward-desc">${reward.desc}</div>
                </div>
                <div class="reward-footer">
                    <span class="reward-cost">${reward.cost} PTS</span>
                    <div>
                        <button class="comic-btn" style="background:#FFF; font-size:11px;" onclick="openEditRewardModal(${reward.id})">?èÔ? Edit</button>
                        <button class="comic-btn btn-add" style="font-size:11px;" onclick="openRedeemModal(${reward.id})">?? Redeem</button>
                    </div>
                </div>
            `;
            container.appendChild(card);
        });
    }

    // ?õÈ? Modal
    function openRedeemModal(rewardId) {
        if (students.length === 0) {
            alert("No students found! Please add students in 'Student Records' first.");
            return;
        }

        currentSelectedReward = rewardsList.find(r => r.id === rewardId);
        document.getElementById('modalRewardTitle').innerText = `${currentSelectedReward.icon} ${currentSelectedReward.title}`;
        document.getElementById('modalRewardCost').innerText = currentSelectedReward.cost;

        const select = document.getElementById('modalStudentSelect');
        select.innerHTML = students.map((s, idx) => `<option value="${idx}">${s.name}</option>`).join('');

        updateModalStudentInfo();
        document.getElementById('redeemModal').style.display = 'flex';
    }

    function updateModalStudentInfo() {
        const select = document.getElementById('modalStudentSelect');
        const idx = select.value;
        if (idx !== null && students[idx]) {
            const currentPoints = students[idx].points;
            const display = document.getElementById('modalStudentScoreDisplay');
            
            if (currentPoints >= currentSelectedReward.cost) {
                display.innerHTML = `Current Points: <span style="color:#06D6A0; font-size:16px; font-weight:bold;">${currentPoints} PTS</span> (Enough!)`;
            } else {
                display.innerHTML = `Current Points: <span style="color:var(--comic-red); font-size:16px; font-weight:bold;">${currentPoints} PTS</span> (Not enough!)`;
            }
        }
    }

    function confirmRedeem() {
        const select = document.getElementById('modalStudentSelect');
        const idx = select.value;
        const student = students[idx];

        if (student.points < currentSelectedReward.cost) {
            alert(`Insufficient points! ${student.name} needs ${currentSelectedReward.cost - student.points} more points.`);
            return;
        }

        if (confirm(`Confirm deducting ${currentSelectedReward.cost} PTS from ${student.name}?`)) {
            student.points -= currentSelectedReward.cost;
            saveStudents();
            renderStudents();
            closeModal('redeemModal');
            alert(`Successfully redeemed! Remaining points for ${student.name}: ${student.points} PTS.`);
        }
    }

    // ‰øÆÊîπ Reward Modal
    function openEditRewardModal(rewardId) {
        currentEditingRewardId = rewardId;
        const item = rewardsList.find(r => r.id === rewardId);
        
        document.getElementById('editRewardTitleInput').value = item.title;
        document.getElementById('editRewardDescInput').value = item.desc;
        document.getElementById('editRewardCostInput').value = item.cost;

        document.getElementById('editRewardModal').style.display = 'flex';
    }

    function saveRewardEdit() {
        const title = document.getElementById('editRewardTitleInput').value.trim();
        const desc = document.getElementById('editRewardDescInput').value.trim();
        const cost = parseInt(document.getElementById('editRewardCostInput').value);

        if (title && !isNaN(cost)) {
            const item = rewardsList.find(r => r.id === currentEditingRewardId);
            item.title = title;
            item.desc = desc;
            item.cost = cost;

            localStorage.setItem('comicRewardsList', JSON.stringify(rewardsList));
            renderRewards();
            closeModal('editRewardModal');
        } else {
            alert("Please fill in valid Title and Cost!");
        }
    }

    function closeModal(modalId) {
        document.getElementById(modalId).style.display = 'none';
    }

    // Initialize Page
    renderGroups();
    renderStudents();
    renderRewards();
